## 이동·문서 링크·목업 브라우저 검사

아래 JavaScript를 임시 `.cjs` 파일로 저장하고 저장소 루트에서 실행한다. 첫 인자는 웹 앱의 절대 경로다. Git 명령은 지정한 저장소에 한해 `safe.directory`를 명시하며 전역 설정을 바꾸지 않는다. 로컬 서버는 빈 포트를 사용하고 검사 후 종료한다. 스크린샷은 임시 스크립트와 같은 위치에 쓴다.

```powershell
node <임시스크립트.cjs> (Resolve-Path 'final_project_cs/frontend/apps/web').Path
git diff --cached --check
```

실제 출력 요약(24개 개별 PASS 행 생략):

```text
Moved file contents: 31/31 unchanged (100%).
Application source, assets, tests and dependencies: unchanged.
Original mockup history: preserved (only parent README link relocated).
Local Markdown file/anchor links: 60/60 passed (100%).
Browser pages and nested mockup navigation: 24/24 passed (100%).
exit 0
```

이동 파일의 Git blob을 기준 커밋과 비교하고, 이전 경로가 남지 않았는지 확인했다. Markdown 링크는 로컬 파일 존재와 앵커를 검사했다. 브라우저는 HTML 12개를 1440×1000·390×844에서 열고, 인트로의 시작 동작·스크롤로 온보딩 iframe을 로드했다. 비교 화면의 iframe 네 개, 본문 표시, 이미지·리소스·JavaScript 오류를 검사했다. `git diff --cached --check`는 출력 없이 exit 0이다.

검증 코드:

```javascript
const fs = require('node:fs');
const path = require('node:path');
const http = require('node:http');
const { execFileSync } = require('node:child_process');
const { createRequire } = require('node:module');
const assert = require('node:assert/strict');
const app = path.resolve(process.argv[2]);
const repo = path.resolve(app, '../../../..');
const relativeApp = 'final_project_cs/frontend/apps/web/';
const base = '882a19b';
const git = (...args) => execFileSync('git', ['-c', 'safe.directory=' + repo.replaceAll('\\', '/'), ...args], { cwd: repo, encoding: 'utf8' }).trim();
const originals = git('ls-tree', '-r', '--name-only', base, relativeApp).split('\n').filter(p => /^final_project_cs\/frontend\/apps\/web\/tripilot-[^/]+$/.test(p));
assert.equal(originals.length, 31);
for (const original of originals) {
  const target = path.join(app, 'mockups', path.basename(original));
  assert.equal(git('hash-object', '--path=' + original, target), git('rev-parse', base + ':' + original), original);
  assert(!fs.existsSync(path.join(repo, original)), original + ' still at old location');
}
console.log('Moved file contents: 31/31 unchanged (100%).');
assert.equal(git('diff', '--name-only', base, '--', relativeApp + 'src', relativeApp + 'public', relativeApp + 'tests', relativeApp + 'package.json', relativeApp + 'package-lock.json'), '');
console.log('Application source, assets, tests and dependencies: unchanged.');
const oldReadme = git('show', base + ':' + relativeApp + 'README.md').replace(/\r\n/g, '\n');
const history = oldReadme.slice(oldReadme.indexOf('## 모바일 온보딩 목업')).replace('(../../README.md)', '(../../../README.md)');
const mockReadme = fs.readFileSync(path.join(app, 'mockups/README.md'), 'utf8').replace(/\r\n/g, '\n');
assert.equal(mockReadme.slice(mockReadme.indexOf('## 모바일 온보딩 목업')).trim(), history.trim());
console.log('Original mockup history: preserved (only parent README link relocated).');
let links = 0;
for (const relative of ['README.md', 'DEVELOPMENT.md', 'mockups/README.md']) {
  const file = path.join(app, relative);
  const content = fs.readFileSync(file, 'utf8');
  for (const match of content.matchAll(/\]\(([^\s)]+)\)/g)) {
    const url = match[1];
    if (/^(?:https?:|mailto:|#)/.test(url)) continue;
    const [target, anchor] = url.split('#');
    const resolved = path.resolve(path.dirname(file), decodeURIComponent(target));
    assert(fs.existsSync(resolved), relative + ' -> ' + url);
    if (anchor && resolved.endsWith('.md')) {
      const headings = [...fs.readFileSync(resolved, 'utf8').matchAll(/^#+\s+(.+)$/gm)].map(m => m[1].trim().toLowerCase().replace(/[^\p{L}\p{N}_\-\s]/gu, '').replace(/\s/g, '-'));
      assert(headings.includes(decodeURIComponent(anchor)), relative + ' missing anchor ' + url);
    }
    links++;
  }
}
console.log('Local Markdown file/anchor links: ' + links + '/' + links + ' passed (100%).');

const { chromium } = createRequire(path.join(app, 'package.json'))('@playwright/test');
const types = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.png': 'image/png', '.svg': 'image/svg+xml' };
const server = http.createServer((req, res) => {
  const requested = decodeURIComponent(new URL(req.url, 'http://127.0.0.1').pathname);
  if (requested === '/favicon.ico') { res.writeHead(204); res.end(); return; }
  const filename = path.resolve(app, '.' + requested);
  if (!filename.startsWith(app + path.sep) || !fs.existsSync(filename) || !fs.statSync(filename).isFile()) { res.writeHead(404); res.end(); return; }
  res.writeHead(200, { 'Content-Type': types[path.extname(filename)] || 'application/octet-stream' });
  fs.createReadStream(filename).pipe(res);
});
(async () => {
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
  const origin = 'http://127.0.0.1:' + server.address().port;
  const browser = await chromium.launch({ channel: 'msedge', headless: true });
  let checks = 0;
  try {
    const html = originals.filter(p => p.endsWith('.html')).map(p => path.basename(p));
    for (const viewport of [{ width: 1440, height: 1000 }, { width: 390, height: 844 }]) {
      const context = await browser.newContext({ viewport, reducedMotion: 'reduce' });
      for (const filename of html) {
        const page = await context.newPage();
        const errors = [];
        page.on('pageerror', e => errors.push(e.message));
        page.on('requestfailed', req => errors.push(req.url() + ': ' + req.failure().errorText));
        page.on('response', response => { if (response.status() >= 400) errors.push(response.status() + ' ' + response.url()); });
        await page.goto(origin + '/mockups/' + filename, { waitUntil: 'networkidle' });
        assert((await page.locator('body').innerText()).trim().length > 10, filename + ' empty body');
        if (/tripilot-intro-(start|cards|expanded|nav-fixed|nav-floating)\.html/.test(filename)) {
          if (await page.locator('#start').count()) await page.locator('#start').click();
          else await page.locator('.inline-card-page').scrollIntoViewIfNeeded();
          await page.waitForFunction(() => [...document.querySelectorAll('iframe')].some(f => f.getAttribute('src') && f.contentDocument?.readyState === 'complete' && f.contentDocument.body?.innerText.trim().length > 20));
          await page.waitForLoadState('networkidle');
        }
        if (filename === 'tripilot-compare.html') assert.equal(page.frames().filter(f => f.url().includes('/mockups/tripilot-intro-')).length, 4);
        const brokenImages = await page.evaluate(() => [...document.images].filter(i => !i.complete || !i.naturalWidth).map(i => i.src));
        assert.deepEqual(brokenImages, [], filename + ' images');
        assert.deepEqual(errors, [], filename + ' browser/resource errors');
        if (filename === 'tripilot-intro-animated-nav-switch.html') await page.screenshot({ path: path.join(__dirname, 'tripilot-mockups-' + viewport.width + '.png') });
        console.log('PASS ' + viewport.width + 'px ' + filename);
        checks++;
        await page.close();
      }
      await context.close();
    }
    console.log('Browser pages and nested mockup navigation: ' + checks + '/24 passed (100%).');
  } finally {
    await browser.close();
    await new Promise(resolve => server.close(resolve));
  }
})().catch(error => { console.error(error); server.close(); process.exitCode = 1; });
```
