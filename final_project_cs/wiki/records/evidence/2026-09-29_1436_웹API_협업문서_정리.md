## 문서·소스 대조

저장소 루트에서 아래 Python 코드를 실행한다. 기존 계약 검사에서 경로를 읽는 함수만 가져와 TypeScript 호출과 Python AST 라우트를 비교한다. `composition.build_domain_routers()` 실행이나 API 통합 시험을 대체하지 않는다.

```python
from pathlib import Path
import ast, json, re, subprocess, sys, unicodedata
from urllib.parse import unquote
sys.stdout.reconfigure(encoding='utf-8')
root = Path.cwd()
def git(*args):
    return subprocess.check_output(['git',*args],stderr=subprocess.DEVNULL).decode('utf-8')
changed = set(git('diff','--name-only','-z').split('\0')) | set(git('ls-files','--others','--exclude-standard','-z').split('\0'))
changed.discard('')
assert all(p.endswith('.md') for p in changed), sorted(changed)
for protected in ['README.md','wiki/delivery/roles.md','wiki/governance/review-policy.md']:
    assert not git('diff','--',protected), protected
policy=(root/'final_project_cs/RULE.md').read_text(encoding='utf-8')
assert '| **최연우·최상욱 공동** | 프론트 연결' in policy
guide=(root/'final_project_cs/wiki/external/web-screen-api.md').read_text(encoding='utf-8')
spec=(root/'final_project_cs/wiki/external/rest-endpoints.md').read_text(encoding='utf-8')
server=ast.parse((root/'final_project_cs/app/modules/travel_ops/trip_api.py').read_text(encoding='utf-8'))
routes=set()
for node in ast.walk(server):
    if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef)):
        for dec in node.decorator_list:
            if isinstance(dec,ast.Call) and isinstance(dec.func,ast.Attribute) and dec.args:
                value=dec.args[0]
                if isinstance(value,ast.Constant) and isinstance(value.value,str) and value.value.startswith('/v1/web/'):
                    routes.add((dec.func.attr.upper(),value.value))
missing=[f'{m} {p}' for m,p in routes if f'{m} {p}' not in spec]
assert not missing, missing
web=root/'final_project_cs/frontend/apps/web'
test=ast.parse((root/'final_project_cs/tests/contract/test_web_client_contract.py').read_text(encoding='utf-8'))
selected=[]
for n in test.body:
    if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id in {'_CALL','_LITERAL','_METHOD'} for t in n.targets): selected.append(n)
    if isinstance(n,ast.FunctionDef) and n.name in {'_code_lines','_call_end','_web_calls'}: selected.append(n)
scope={'re':re,'LIVE':web/'src/lib/live'}
exec(compile(ast.Module(body=selected,type_ignores=[]),'<existing-source-extractor>','exec'),scope)
calls=scope['_web_calls']()
normalized={(m,re.sub(r'\{[^}]*\}','{}',p)) for m,p in routes}
assert not calls-normalized, sorted(calls-normalized)
pages=[]
for p in sorted((web/'src/app').rglob('page.tsx')):
    route='/'+p.parent.relative_to(web/'src/app').as_posix()
    route=route.replace('/.','/')
    pages.append(route)
assert all(f'`{p}`' in guide for p in pages), pages
def plain(text):
    return re.sub(r'^```[^\n]*\n.*?^```[^\n]*$', '', text, flags=re.M|re.S)
def anchors(path):
    text=plain(path.read_text(encoding='utf-8'))
    result=set(re.findall(r'<a\s+(?:id|name)=["\']([^"\']+)',text))
    counts={}
    for heading in re.findall(r'^#{1,6}\s+(.+?)\s*#*$',text,re.M):
        heading=re.sub(r'\[([^\]]*)\]\([^)]*\)',r'\1',heading)
        slug=''.join(c for c in heading.lower() if c.isalnum() or c in '_ -')
        slug=slug.replace(' ','-')
        number=counts.get(slug,0);counts[slug]=number+1
        result.add(slug+(f'-{number}' if number else ''))
    return result
checks=0; failures=[]; preexisting=[]
for relative in sorted(changed):
    p=root/relative
    if not p.is_file(): continue
    body=plain(p.read_text(encoding='utf-8'))
    try: old=git('show',f'HEAD:{relative}')
    except subprocess.CalledProcessError: old=''
    for url in re.findall(r'(?<!!)\[[^\]\n]*\]\(([^\s)]+)(?:\s+"[^"]*")?\)',body):
        if re.match(r'[a-zA-Z][a-zA-Z0-9+.-]*:',url): continue
        target,_,anchor=unquote(url.strip('<>')).partition('#')
        dest=(p.parent/target).resolve() if target else p
        issue=None
        if not dest.exists(): issue='missing file'
        elif anchor and dest.suffix.lower()=='.md' and anchor not in anchors(dest): issue='missing anchor'
        checks+=1
        if issue:
            entry={'file':relative,'target':url,'issue':issue}
            (preexisting if f']({url})' in old else failures).append(entry)
report={'changed_markdown':len(changed),'documented_web_routes':f'{len(routes)-len(missing)}/{len(routes)}',
        'frontend_calls_match_source_routes':f'{len(calls-(calls-normalized))}/{len(calls)}',
        'screen_routes_in_guide':f'{len(pages)}/{len(pages)}','local_link_checks':checks,
        'preexisting_link_failures':preexisting,'new_link_failures':failures,
        'role_documents_unchanged':True,'frontend_connection_is_joint':True,
        'api_spec_lines':len(spec.splitlines()),'guide_lines':len(guide.splitlines())}
print(json.dumps(report,ensure_ascii=False,indent=2))
sys.exit(bool(failures))
```
